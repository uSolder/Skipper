# uSolder Skipper (STM32H523ZET6)
skipper.name=uSolder Skipper

skipper.build.core=STMicroelectronics:arduino
skipper.build.board=SKIPPER
skipper.build.variant=SKIPPER
skipper.build.variant_h=variant_SKIPPER.h
skipper.build.series=STM32H5xx
skipper.build.product_line=STM32H523xx
skipper.build.mcu=cortex-m33
skipper.build.fpu=-mfpu=fpv5-sp-d16
skipper.build.float-abi=-mfloat-abi=hard
skipper.build.flash_offset=0x0
skipper.build.st_extra_flags=-D{build.product_line} {build.enable_usb} {build.xSerial}
skipper.build.extra_flags='-DUSB_MANUFACTURER_STRING="uSolder"' '-DUSB_PRODUCT_STRING="Skipper"'

skipper.upload.maximum_size=524288
skipper.upload.maximum_data_size=278528

# USB CDC/HID on the USB-C DRP connector
skipper.vid.0=0x0483
skipper.pid.0=0x5740

skipper.debug.svd_file={runtime.tools.STM32_SVD.path}/svd/STM32H5xx/STM32H523.svd

# Upload menu
# The on-board DAPLink probe shows up as a USB drive on the USB-C UFP connector.
skipper.menu.upload_method.dapMassStorage=DAPLink (drag and drop)
skipper.menu.upload_method.dapMassStorage.upload.protocol=
skipper.menu.upload_method.dapMassStorage.upload.tool=STMicroelectronics:massStorageCopy
skipper.menu.upload_method.dapMassStorage.node=SKIPPER,DAPLINK

# ROM bootloader over the DAPLink virtual COM port (USART3): hold BOOT, press RESET.
skipper.menu.upload_method.serialMethod=STM32CubeProgrammer (Serial, BOOT button)
skipper.menu.upload_method.serialMethod.upload.protocol=serial
skipper.menu.upload_method.serialMethod.upload.options=-c {serial.port.file} -a {upload.address} -s {upload.start} -p {upload.parity}
skipper.menu.upload_method.serialMethod.upload.tool=STMicroelectronics:stm32CubeProg

# ROM bootloader over USB on the USB-C DRP connector: hold BOOT, press RESET.
skipper.menu.upload_method.dfuMethod=STM32CubeProgrammer (DFU, BOOT button)
skipper.menu.upload_method.dfuMethod.upload.protocol=dfu
skipper.menu.upload_method.dfuMethod.upload.options=-v {upload.vid} -p {upload.pid} -a {upload.address} -s {upload.start}
skipper.menu.upload_method.dfuMethod.upload.tool=STMicroelectronics:stm32CubeProg
